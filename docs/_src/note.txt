# 登录MySQL
mysql -u root -p

# 创建数据库
CREATE DATABASE IF NOT EXISTS ailabre 
DEFAULT CHARACTER SET utf8mb4 
COLLATE utf8mb4_0900_ai_ci;

# 导入项目提供的初始化脚本
USE ailabre;
SOURCE /path/to/ailabre/ailabre.sql;







# 克隆项目代码
git clone https://github.com/your-team/ailabre.git
cd ailabre

# 安装前端依赖
cd frontend
npm install

# 安装后端AI服务依赖
cd ../ai_service
python -m venv venv
source venv/bin/activate  # Windows执行: venv\Scripts\activate
pip install -r requirements.txt







# backend/src/main/resources/application.yml
spring:
  datasource:
    url: jdbc:mysql://localhost:3306/ailabre?useUnicode=true&characterEncoding=utf8mb4
    username: root
    password: your_mysql_password
  redis:
    host: localhost
    port: 6379

ai:
  api-key: your_api_key_here
  model-name: ProximaAI/Proxima-7B-Instruct
  base-url: https://api.siliconflow.cn/v1





# 启动Redis（如未启动）
redis-server

# 启动Spring Boot后端服务
cd backend
mvn spring-boot:run

# 启动AI服务（新终端窗口）
cd ai_service
source venv/bin/activate  # Windows执行: venv\Scripts\activate
uvicorn main:app --host 0.0.0.0 --port 8000
